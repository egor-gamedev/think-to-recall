import numpy as np
res = pd.read_csv("results/deseq2_fasting_vs_control.tsv", sep="\t").dropna(subset=["padj"])
res["neglog10"] = -np.log10(res["padj"])
sig = (res["padj"] < 0.05) & (res["log2FoldChange"].abs() > 1)
up = sig & (res["log2FoldChange"] > 0)
down = sig & (res["log2FoldChange"] < 0)

fig, ax = plt.subplots(figsize=(6, 5))

ax.scatter(res.loc[down, "log2FoldChange"], res.loc[down, "neglog10"], s=8, c="#2c7fb8")
ax.axhline(-np.log10(0.05), ls="--", lw=0.8, c="k")
ax.axvline(-1, ls="--", lw=0.8, c="k"); ax.axvline(1, ls="--", lw=0.8, c="k")
ax.set_xlabel("log2 fold change"); ax.set_ylabel("-log10 FDR")
for _, r in res[sig].nsmallest(10, "padj").iterrows():
